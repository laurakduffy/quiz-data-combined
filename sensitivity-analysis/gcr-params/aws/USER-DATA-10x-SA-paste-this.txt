#!/usr/bin/env bash
# 10x run (10,000,000 samples). Use on a big instance: r7i.8xlarge (32 vCPU, 256 GB)
# or m7i.8xlarge (32 vCPU, 128 GB). Requires a vCPU-limit increase to >=32.
BUCKET="gcr-sa-2026"
NSAMPLES="10000000"
NBATCHES="100"          # keeps ~100k samples/batch -> ~3.8 GB per process
CONCURRENCY="24"        # peak procs = 24+1; ~5 GB each -> ~125 GB (fine on 256 GB)
BUNDLE_KEY="gcr-aws-bundle.tar.gz"
RESULTS_KEY="gcr-results-10x.tar.gz"
LOG_KEY="gcr-run-10x.log"

set -xeuo pipefail
exec > /var/log/gcr-run.log 2>&1

dnf install -y unzip tar gzip >/dev/null 2>&1 || yum install -y unzip tar gzip
curl -s "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o /tmp/awscliv2.zip
unzip -q /tmp/awscliv2.zip -d /tmp
/tmp/aws/install
AWS=/usr/local/bin/aws

cd /root
"$AWS" s3 cp "s3://$BUCKET/$BUNDLE_KEY" ./bundle.tar.gz
tar xzf bundle.tar.gz
find quiz-demo -name '*.sh' -exec sed -i 's/\r$//' {} +

NSAMPLES="$NSAMPLES" NBATCHES="$NBATCHES" CONCURRENCY="$CONCURRENCY" \
  bash quiz-demo/sensitivity-analysis/gcr-params/aws/run_on_ec2.sh || true

"$AWS" s3 cp /tmp/gcr-results.tar.gz "s3://$BUCKET/$RESULTS_KEY" || true
"$AWS" s3 cp /var/log/gcr-run.log    "s3://$BUCKET/$LOG_KEY"     || true

shutdown -h now
